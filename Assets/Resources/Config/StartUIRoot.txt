using UnityEngine;
using UnityEngine.UI;
<rep>useDOtween</rep>
public class StartUIRoot : MonoBehaviour
{
    public Image StartPanelBG;
    public Button StartBtn;
    private void OnEnable()
    {
        EventCenter.AddListener<GameStartEvent>(OnGameStartEvent);
    }
    void Start()
    {
        StartPanelBG = GameObject.Find("StartPanel_BG").GetComponent<Image>();
        StartBtn = GameObject.Find("StartPanel_StartBtn").GetComponent<Button>();
        uiManager.INSTANCE.BtnExtra(StartBtn, () => EventCenter.TriggerEvent(new GameStartEvent { }));
        StartPanelBG.sprite = Resources.Load<Sprite>("<rep>StartUIRoot_StartPanelBGPath</rep>");
        StartBtn.gameObject.GetComponent<Image>().sprite = Resources.Load<Sprite>("<rep>StartUIRoot_StartPanelBtnPath</rep>");
    }
    void OnGameStartEvent(GameStartEvent evt)
    {
        EventCenter.TriggerEvent(new SetPlayerActState{targetState = true});
        <rep>StartUIRoot_HidePanel</rep>
        <rep>StartUIRoot_useDOtween_HidePanel</rep>
        
    }
    private void OnDisable()
    {
        EventCenter.RemoveListener<GameStartEvent>(OnGameStartEvent);
    }
}
